## JavaScript

```js
function initLogoStackLoader() {
  const root = document.querySelector("[data-logo-loader-init]");
  if (!root) return () => {};

  const logoDuplicateCount = 5; // copies of the logo made behind the original

  const bg = root.querySelector("[data-logo-loader-bg]");
  const stack = root.querySelector("[data-logo-loader-stack]");
  const logo = root.querySelector("[data-logo-loader-logo]");
  const bar = root.querySelector("[data-logo-loader-bar]");
  const track = root.querySelector("[data-logo-loader-track]") || bar;
  if (!bg || !stack || !logo) return () => {};

  const resetTargets = [...root.querySelectorAll("[data-logo-loader-reset]")].filter((element) => element !== stack);
  const reduceMotion = window.matchMedia("(prefers-reduced-motion: reduce)").matches;

  const logoDuplicates = [];
  for (let i = 0; i < logoDuplicateCount; i += 1) {
    const duplicate = logo.cloneNode(true);
    duplicate.setAttribute("aria-hidden", "true");
    stack.appendChild(duplicate);
    logoDuplicates.push(duplicate);
  }

  const em = parseFloat(getComputedStyle(stack).fontSize);

  const timeline = gsap.timeline({
    defaults: {
      duration: 0.4,
      ease: "power2.out",
    },
  });

  gsap.set(logoDuplicates, {
    opacity: 0,
    z: 0,
  });

  if (resetTargets.length) {
    timeline.set(resetTargets, {
      autoAlpha: 1,
    }, 0);
  }

  timeline.to(stack, {
    autoAlpha: 1,
    duration: 0.5,
  });

  if (reduceMotion) {
    timeline
      .addLabel("impact", "+=0.6")
      .to(root, {
        autoAlpha: 0,
        duration: 0.5,
        ease: "power2.inOut",
      }, "impact+=0.5");
  } else {
    timeline
      .addLabel("tilt", "-=0.1")
      .to(stack, {
        rotationY: -40,
        rotationX: 20,
        xPercent: -15,
        duration: 0.6,
        ease: "power4.out",
      }, "tilt")
      .to(logoDuplicates, {
        z: (i) => -(i + 1) * 2.5 * em,
        opacity: (i) => Math.pow(0.5, i + 1),
        duration: 0.55,
        ease: "power4.out",
        stagger: {
          each: 0.05,
          ease: "power1.out",
        },
      }, "tilt+=0.1")
      .addLabel("smash", "+=0.4")
      .to(stack, {
        rotationY: 0,
        rotationX: 0,
        xPercent: 0,
        ease: "power4.in",
      }, "smash")
      .to(logoDuplicates, {
        z: 0,
        ease: "power4.in",
      }, "smash")
      .addLabel("impact")
      .set(logoDuplicates, {
        opacity: 0,
      }, "impact")
      .to(stack, {
        scale: 1.05,
        duration: 0.1,
      }, "impact")
      .to(stack, {
        scale: 1,
        duration: 0.6,
        ease: "elastic.out(1, 0.5)",
      })
      .to(stack, {
        autoAlpha: 0,
      }, "impact+=0.5")
      .to(bg, {
        scaleY: 0,
        duration: 0.5,
        ease: "expo.inOut",
      }, "impact+=0.6");
  }

  if (bar) {
    timeline
      .to(bar, {
        scaleX: 1,
        duration: timeline.labels.impact,
        ease: "power2.inOut",
      }, 0)
      .to(track, {
        autoAlpha: 0,
      }, "impact+=0.5");
  }

  timeline.set(root, {
    display: "none",
  });

  return () => {
    timeline.kill();
    logoDuplicates.forEach((duplicate) => duplicate.remove());
    gsap.set([root, bg, stack, logo, bar, track].filter(Boolean), {
      clearProps: "all",
    });
  };
}

// Initialize Logo Stack Loader
document.addEventListener("DOMContentLoaded", () => {
  initLogoStackLoader();
});
```

## Documentation for Resource

### Change logo

Swap the SVG inside the element marked `[data-logo-loader-logo]` for your own, an inline SVG or an image both work. That's the only logo you place. The script duplicates it inside `[data-logo-loader-stack]` to build the stack.

### Amount of duplicates

`logoDuplicateCount` at the top of the script sets how many copies sit behind your logo, 5 by default. Each copy sits 2.5em further back and is half as visible as the one in front of it. Both of those live in the tween that fans them out, so tweak the depth and the opacity there.

### Tilt

The stack rotates -40 degrees on the Y axis and 20 on the X axis, and moves 15% to the left so the fanned out copies stay centred on the page. Those are the three values in the `tilt` tween. The `perspective` on `.logo-loader` sets how dramatic the depth looks, a lower value means more.

### Background

The background is its own element with `[data-logo-loader-bg]`, so we can animate it without moving the logo. At the end we scale it to 0 on the Y axis from the top, which reveals your page from the bottom up. Want it to slide or fade instead? That's the last tween in the timeline.

### Progress bar

This one is optional. Add an element with `[data-logo-loader-track]` and inside it a div with `[data-logo-loader-bar]`. The bar scales from 0 to 1 on the X axis and reaches the end exactly when the stack snaps back into one logo. Don't want a bar? Leave both elements out, the script checks for them.

### Prevent FOUC

Anything with `[data-logo-loader-reset]` starts at opacity 0 through the custom CSS, so nothing flashes before GSAP kicks in. The stack has it by default. Add it to anything else inside your loader that should only show up when the timeline starts, and the script sets those elements back to visible as its very first step.

### Reduced motion

If someone has reduced motion turned on, the stack doesn't tilt or fan out. The logo fades in, holds for a moment, and the whole loader fades out.

### Webflow

On the canvas the loader would cover everything, so the custom CSS hides it in the Designer with `.wf-design-mode .logo-loader { display: none; }`. Comment that rule out while you style the loader and put it back when you're done. It only applies inside the Designer.
